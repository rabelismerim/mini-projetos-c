cmake_minimum_required(VERSION 3.16)
project(miniprojetos_c LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
file(GLOB_RECURSE PROJECT_SOURCES CONFIGURE_DEPENDS "projetos/*/main.c")
foreach(SOURCE IN LISTS PROJECT_SOURCES)
    get_filename_component(DIRECTORY "${SOURCE}" DIRECTORY)
    get_filename_component(NAME "${DIRECTORY}" NAME)
    add_executable(${NAME} "${SOURCE}")
    target_include_directories(${NAME} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
    if(MSVC)
        target_compile_options(${NAME} PRIVATE /W4 /WX /utf-8)
        target_compile_definitions(${NAME} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${NAME} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
